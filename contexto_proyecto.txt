#!/usr/bin/env python3
"""
Script para generar archivo de contexto del proyecto Django
Uso: python generar_contexto.py
"""

import os
from pathlib import Path

# Extensiones de archivos a incluir
EXTENSIONES = {'.py', '.html', '.css', '.js', '.json', '.md', '.txt', '.yml', '.yaml'}

# Carpetas a ignorar
IGNORAR_CARPETAS = {
    'venv', '.venv', 'env', '.env',
    '__pycache__', '.git', '.github',
    'staticfiles', 'media', 'node_modules',
    '.pytest_cache', '.mypy_cache', '.ruff_cache',
    'dist', 'build', '.eggs'
}

# Archivos a ignorar
IGNORAR_ARCHIVOS = {
    '.gitignore', '.env', '.env.local',
    'db.sqlite3', '*.pyc', '*.pyo',
    'contexto_proyecto.txt'
}

def debe_incluir(ruta: Path) -> bool:
    """Determina si un archivo debe ser incluido"""
    # Verificar extensión
    if ruta.suffix.lower() not in EXTENSIONES:
        return False
    
    # Verificar si está en carpetas ignoradas
    for parte in ruta.parts:
        if parte in IGNORAR_CARPETAS:
            return False
    
    # Verificar nombre de archivo
    if ruta.name in IGNORAR_ARCHIVOS:
        return False
    
    # Verificar patrones glob
    for patron in IGNORAR_ARCHIVOS:
        if '*' in patron and ruta.match(patron):
            return False
    
    return True

def generar_contexto(directorio_raiz: str = '.'):
    """Genera archivo de contexto con todo el código del proyecto"""
    
    ruta_raiz = Path(directorio_raiz).resolve()
    archivos_incluidos = []
    
    print(f"🔍 Escaneando proyecto en: {ruta_raiz}")
    print("=" * 60)
    
    # Recorrer todos los archivos
    for ruta_archivo in ruta_raiz.rglob('*'):
        if ruta_archivo.is_file() and debe_incluir(ruta_archivo):
            try:
                # Leer contenido del archivo
                with open(ruta_archivo, 'r', encoding='utf-8') as f:
                    contenido = f.read()
                
                # Calcular ruta relativa
                ruta_relativa = ruta_archivo.relative_to(ruta_raiz)
                
                archivos_incluidos.append({
                    'ruta': str(ruta_relativa),
                    'contenido': contenido,
                    'lineas': len(contenido.splitlines())
                })
                
                print(f"✅ {ruta_relativa} ({len(contenido.splitlines())} líneas)")
                
            except Exception as e:
                print(f"⚠️  Error leyendo {ruta_archivo}: {e}")
    
    # Ordenar archivos por ruta
    archivos_incluidos.sort(key=lambda x: x['ruta'])
    
    # Generar archivo de salida
    archivo_salida = ruta_raiz / 'contexto_proyecto.txt'
    
    with open(archivo_salida, 'w', encoding='utf-8') as f:
        f.write("=" * 80 + "\n")
        f.write("CONTEXTO DEL PROYECTO DJANGO\n")
        f.write(f"Generado automáticamente desde: {ruta_raiz}\n")
        f.write(f"Total de archivos: {len(archivos_incluidos)}\n")
        f.write("=" * 80 + "\n\n")
        
        # Índice de archivos
        f.write("ÍNDICE DE ARCHIVOS:\n")
        f.write("-" * 80 + "\n")
        for i, archivo in enumerate(archivos_incluidos, 1):
            f.write(f"{i:3d}. {archivo['ruta']} ({archivo['lineas']} líneas)\n")
        f.write("\n" + "=" * 80 + "\n\n")
        
        # Contenido de cada archivo
        for archivo in archivos_incluidos:
            f.write("=" * 80 + "\n")
            f.write(f"ARCHIVO: {archivo['ruta']}\n")
            f.write("=" * 80 + "\n")
            f.write(archivo['contenido'])
            f.write("\n\n")
    
    print("\n" + "=" * 60)
    print(f"✅ Contexto generado exitosamente!")
    print(f"📄 Archivo: {archivo_salida}")
    print(f"📊 Total: {len(archivos_incluidos)} archivos")
    print(f"📏 Tamaño: {archivo_salida.stat().st_size / 1024:.2f} KB")
    print("=" * 60)
    print("\n📋 PRÓXIMOS PASOS:")
    print("1. Abre el archivo 'contexto_proyecto.txt'")
    print("2. Copia TODO el contenido")
    print("3. Pégalo en el chat de Claude")
    print("4. Claude tendrá contexto completo de tu proyecto")

if __name__ == '__main__':
    generar_contexto()

# news/context_processors.py

from .models import SiteConfig

def site_config(request):
    """Configuración del sitio con soporte de temas"""
    try:
        config = SiteConfig.objects.first()
        
        if config:
            # Detectar tema activo (desde cookie, header o default)
            theme = request.COOKIES.get('theme_mode', 'light')
            
            if theme == 'dark':
                primary = config.dark_primary_color
                secondary = config.dark_secondary_color
                background = config.dark_background_color
                text = config.dark_text_color
            else:
                primary = config.light_primary_color
                secondary = config.light_secondary_color
                background = config.light_background_color
                text = config.light_text_color
            
            return {
                'site_config': config,
                'site_name': config.site_name,
                'primary_color': primary,
                'secondary_color': secondary,
                'background_color': background,
                'text_color': text,
                'current_theme': theme,
                # Pasar todos los colores para el panel de personalización
                'light_primary': config.light_primary_color,
                'light_secondary': config.light_secondary_color,
                'light_background': config.light_background_color,
                'light_text': config.light_text_color,
                'dark_primary': config.dark_primary_color,
                'dark_secondary': config.dark_secondary_color,
                'dark_background': config.dark_background_color,
                'dark_text': config.dark_text_color,
            }
        else:
            return {
                'site_config': None,
                'site_name': 'Periódico Digital',
                'primary_color': '#607D8B',
                'secondary_color': '#FF9800',
                'background_color': '#F8F9FA',
                'text_color': '#212529',
                'current_theme': 'light',
            }
    except Exception as e:
        return {
            'site_config': None,
            'site_name': 'Periódico Digital',
            'primary_color': '#607D8B',
            'secondary_color': '#FF9800',
            'background_color': '#F8F9FA',
            'text_color': '#212529',
            'current_theme': 'light',
        }

from django.shortcuts import render
from django.http import HttpResponse
from django.template import TemplateDoesNotExist, TemplateSyntaxError
from .models import Post
import traceback

def debug_post_detail(request, post_id):
    """Vista de diagnóstico para encontrar errores en post_detail.html"""
    try:
        post = Post.objects.get(id=post_id)
        
        # Intentar renderizar el template
        context = {
            'post': post,
            'can_edit': True,
            'can_delete': True,
        }
        
        return render(request, 'post_detail.html', context)
        
    except Post.DoesNotExist:
        return HttpResponse(f"""
            <h2>❌ Post no encontrado</h2>
            <p>No existe un post con ID {post_id}</p>
        """, status=404)
        
    except TemplateDoesNotExist as e:
        return HttpResponse(f"""
            <h2>❌ Template no encontrado</h2>
            <p><strong>Error:</strong> {e}</p>
            <p>Verifica que el archivo <code>templates/post_detail.html</code> exista.</p>
        """, status=404)
        
    except TemplateSyntaxError as e:
        return HttpResponse(f"""
            <h2>❌ Error de sintaxis en template</h2>
            <p><strong>Error:</strong> {e}</p>
            <pre style="background:#fee; padding:20px; border:2px solid #c00;">
{traceback.format_exc()}
            </pre>
        """, status=500)
        
    except Exception as e:
        return HttpResponse(f"""
            <h2>❌ Error inesperado</h2>
            <p><strong>Tipo:</strong> {type(e).__name__}</p>
            <p><strong>Mensaje:</strong> {e}</p>
            <pre style="background:#fee; padding:20px; border:2px solid #c00;">
{traceback.format_exc()}
            </pre>
        """, status=500)

class Category(models.Model):
    name = models.CharField(max_length=100, unique=True)
    slug = models.SlugField(max_length=100, unique=True, blank=True)
    description = models.TextField(blank=True, help_text="Descripción opcional de la categoría")
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        verbose_name_plural = "Categories"
        ordering = ['name']

    def __str__(self):
        return self.name

    def save(self, *args, **kwargs):
        if not self.slug:
            from django.utils.text import slugify
            self.slug = slugify(self.name)
        super().save(*args, **kwargs)

# news/permissions.py
from rest_framework import permissions

class IsAuthorOrEditor(permissions.BasePermission):
    """
    Permiso personalizado: 
    - Cualquiera puede leer (GET, HEAD, OPTIONS).
    - Solo el autor o usuarios con permisos de 'change_post' o 'delete_post' pueden editar/borrar.
    """
    def has_object_permission(self, request, view, obj):
        # Métodos seguros (lectura) siempre permitidos
        if request.method in permissions.SAFE_METHODS:
            return True

        # Métodos de escritura (PUT, PATCH, DELETE) requieren se el autor o tener permisos
        return(
            obj.author == request.user or
            request.user.has_perm('news.charge_post') or
            request.user.has_perm('news.delete_post') or
            request.user.is_superuser
        )

class UserRegistrationSerializer(serializers.ModelSerializer):
    password = serializers.CharField(write_only=True, min_length=8)
    password_confirm = serializers.CharField(write_only=True)

    class Meta:
        model = User
        fields = ['username', 'email', 'password', 'password_confirm']

    def validate(self, data):
        if data['password'] != data['password_confirm']:
            raise serializers.ValidationError("Las contraseñas no coinciden")
        return data

    def create(self, validated_data):
        validated_data.pop('password_confirm')
        user = User.objects.create_user(
            username=validated_data['username'],
            email=validated_data.get('email', ''),
            password=validated_data['password']
        )
        from django.contrib.auth.models import Group
        readers_group, created = Group.objects.get_or_create(name='readers')
        user.groups.add(readers_group)
        return user

class NotificationSerializer(serializers.ModelSerializer):
    sender_username = serializers.ReadOnlyField(source='sender.username')
    post_title = serializers.ReadOnlyField(source='post.title')
    comment_id = serializers.SerializerMethodField()

    class Meta:
        model = Notification
        fields = [
            'id', 'recipient', 'sender', 'sender_username', 
            'post', 'post_title', 'notification_type', 
            'message', 'is_read', 'created_at', 'comment_id'
        ]
        read_only_fields = ['id', 'recipient', 'sender', 'created_at']

    def get_comment_id(self, obj):
        try:
            # Solo buscamos comentario si es de tipo 'reply' o 'comment' y hay un post
            if obj.notification_type in ['reply', 'comment'] and obj.post:
                # Buscamos el comentario más reciente de este autor en este post
                comment = Comment.objects.filter(
                    post=obj.post,
                    author=obj.sender
                ).order_by('-created_at').first()
                
                if comment:
                    return comment.id
        except Exception as e:
            # Si algo falla, lo registramos pero no rompemos toda la API
            print(f"⚠️ Error obteniendo comment_id para notificación {obj.id}: {e}")
            
        return None

from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.contrib.admin.views.decorators import staff_member_required
from django.core.files.storage import default_storage
from django.core.files.base import ContentFile
import os
import uuid

@csrf_exempt
@staff_member_required
def tinymce_upload_image(request):
    """Maneja la subida de imágenes desde TinyMCE"""
    if request.method == 'POST' and request.FILES.get('file'):
        image = request.FILES['file']
        
        # Validar tipo de archivo
        allowed_types = ['image/jpeg', 'image/png', 'image/gif', 'image/webp']
        if image.content_type not in allowed_types:
            return JsonResponse({'error': 'Tipo de archivo no permitido'}, status=400)
        
        # Validar tamaño (máximo 5MB)
        if image.size > 5 * 1024 * 1024:
            return JsonResponse({'error': 'La imagen es muy grande (máx 5MB)'}, status=400)
        
        # Generar nombre único
        ext = os.path.splitext(image.name)[1]
        filename = f"tinymce/{uuid.uuid4().hex}{ext}"
        
        # Guardar archivo
        file_path = default_storage.save(filename, ContentFile(image.read()))
        file_url = default_storage.url(file_path)
        
        return JsonResponse({'location': file_url})
    
    return JsonResponse({'error': 'No se proporcionó archivo'}, status=400)

class HomePageView(ListView):
    model = Post
    template_name = 'index.html'
    context_object_name = 'all_posts'
    paginate_by = 10

    def get_queryset(self):
        queryset = super().get_queryset()
        query = self.request.GET.get('q')
        if query:
            queryset = queryset.filter(
                Q(text__icontains=query) | Q(author__username__icontains=query)
            )
        return queryset
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context['query'] = self.request.GET.get('q', '')
        current_page = context['page_obj'].number
        total_pages = context['page_obj'].paginator.num_pages

        if total_pages <= 7:
            context['page_range'] = context['page_obj'].paginator.page_range
        else:
            start = max(1, current_page - 2)
            end = min(total_pages, current_page + 2)
            context['page_range'] = range(start, end + 1)
        return context

class PostDetailAPIView(generics.RetrieveUpdateDestroyAPIView):
    queryset = Post.objects.all()
    serializer_class = PostSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly, IsAuthorOrEditor]
    authentication_classes = [TokenAuthentication]
    parser_classes = [MultiPartParser, FormParser, JSONParser]

    def perform_update(self, serializer):
        if not self.get_object_permissions():
            raise PermissionDenied("No tienes permisos para editar este artículo.")
        serializer.save()

    def get_object_permissions(self):
        obj = self.get_object()
        user = self.request.user
        if user.is_superuser:
            return True
        if obj.author == user:
            return True
        if user.has_perm('news.change_post'):
            return True
        return False

    def update(self, request, *args, **kwargs):
        partial = kwargs.pop('partial', False)
        instance = self.get_object()
        serializer = self.get_serializer(instance, data=request.data, partial=partial)
        serializer.is_valid(raise_exception=True)
        self.perform_update(serializer)
        return Response(serializer.data)

    def partial_update(self, request, *args, **kwargs):
        kwargs['partial'] = True
        return self.update(request, *args, **kwargs)

    def perform_destroy(self, instance):
        obj = self.get_object()
        user = self.request.user
        if not (user.is_superuser or obj.author == user or user.has_perm('news.delete_post')):
            raise PermissionDenied("No tienes permisos para borrar este artículo.")
        instance.delete()

class CommentListAPIView(generics.ListCreateAPIView):
    serializer_class = CommentSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly]
    authentication_classes = [TokenAuthentication]

    def get_queryset(self):
        post_id = self.kwargs.get('post_id')
        return Comment.objects.filter(
            post_id=post_id, 
            parent__isnull=True
        ).select_related('author').prefetch_related(
            'replies__author',
            'replies__replies__author',
            'replies__replies__replies__author',
        ).order_by('created_at')

    def create(self, request, *args, **kwargs):
        post_id = self.kwargs.get('post_id')
        
        try:
            post = Post.objects.get(id=post_id)
        except Post.DoesNotExist:
            raise NotFound("El artículo no existe")
        
        if not post.allow_comments:
            raise PermissionDenied("Los comentarios están deshabilitados")
        
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        
        parent = serializer.validated_data.get('parent')
        
        if parent and parent.post_id != post.id:
            raise ValidationError("El comentario padre no pertenece a este artículo")
        
        comment = Comment.objects.create(
            post=post,
            author=request.user,
            parent=parent,
            text=serializer.validated_data['text']
        )
        
        # Crear notificación
        if parent:
            if parent.author != request.user:
                Notification.objects.create(
                    recipient=parent.author,
                    sender=request.user,
                    post=post,
                    notification_type='reply',
                    message=f'{request.user.username} respondió a tu comentario en "{post.title}"'
                )
        else:
            if post.author != request.user:
                Notification.objects.create(
                    recipient=post.author,
                    sender=request.user,
                    post=post,
                    notification_type='comment',
                    message=f'{request.user.username} comentó en tu artículo "{post.title}"'
                )
        
        output_serializer = self.get_serializer(comment)
        return Response(output_serializer.data, status=201)

class PostImageListAPIView(generics.ListCreateAPIView):
    serializer_class = PostImageSerializer
    permission_classes = [permissions.IsAuthenticatedOrReadOnly]
    authentication_classes = [TokenAuthentication]

    def get_queryset(self):
        post_id = self.kwargs.get('post_id')
        return PostImage.objects.filter(post_id=post_id)

    def perform_create(self, serializer):
        try:
            post_id = self.kwargs.get('post_id')
            post = Post.objects.get(id=post_id)
            
            if not (self.request.user.is_superuser or 
                    post.author == self.request.user or 
                    self.request.user.has_perm('news.change_post')):
                raise PermissionDenied("No tienes permiso para agregar imágenes a este artículo")
            
            serializer.save(post=post)
            
        except Post.DoesNotExist:
            raise NotFound("El artículo no existe")
        except Exception as e:
            import traceback
            error_detail = traceback.format_exc()
            print(f"❌ ERROR AL SUBIR IMAGEN: {error_detail}")
            raise APIException(f"Error al procesar la imagen: {str(e)}")

    def create(self, request, *args, **kwargs):
        try:
            return super().create(request, *args, **kwargs)
        except Exception as e:
            import traceback
            error_detail = traceback.format_exc()

class PostImageReorderAPIView(APIView):
    permission_classes = [permissions.IsAuthenticated]
    authentication_classes = [TokenAuthentication]

    def post(self, request, post_id):
        try:
            post = Post.objects.get(id=post_id)
            
            if not (request.user.is_superuser or 
                    post.author == request.user or 
                    request.user.has_perm('news.change_post')):
                return Response({'error': 'No tienes permiso'}, status=403)
            
            images_data = request.data.get('images', [])
            
            for img_data in images_data:
                image_id = img_data.get('id')
                new_order = img_data.get('order')
                
                if image_id and new_order is not None:
                    PostImage.objects.filter(id=image_id, post=post).update(order=new_order)
            
            return Response({'success': True, 'message': 'Imágenes reordenadas'})
        except Post.DoesNotExist:
            return Response({'error': 'Artículo no encontrado'}, status=404)

class LikeToggleAPIView(APIView):
    permission_classes = [permissions.IsAuthenticated]
    authentication_classes = [TokenAuthentication]

    def post(self, request, post_id):
        try:
            post = Post.objects.get(id=post_id)
        except Post.DoesNotExist:
            return Response({'error': 'Artículo no encontrado'}, status=404)

        existing_like = Like.objects.filter(post=post, user=request.user).first()

        if existing_like:
            existing_like.delete()
            return Response({
                'liked': False,
                'like_count': post.likes.count(),
                'message': 'Like eliminado'
            })
        else:
            Like.objects.create(post=post, user=request.user)

            if post.author != request.user:
                Notification.objects.create(
                    recipient=post.author,
                    sender=request.user,
                    post=post,
                    notification_type='like',
                    message=f'{request.user.username} dio like a tu artículo "{post.title}"'
                )
                
            return Response({
                'liked': True,
                'like_count': post.likes.count(),
                'message': 'Like agregado'
            })

@staff_member_required
def visual_editor_view(request):
    """Editor visual tipo Elementor con filtros por scope"""
    scope_filter = request.GET.get('scope', 'all')
    
    if scope_filter == 'all':
        styles = CustomStyle.objects.all()
    else:
        styles = CustomStyle.objects.filter(scope=scope_filter)
    
    # Agrupar por scope y categoría
    grouped = {}
    for style in styles:
        if style.scope not in grouped:
            grouped[style.scope] = {}
        if style.category not in grouped[style.scope]:
            grouped[style.scope][style.category] = []
        
        grouped[style.scope][style.category].append({
            'id': style.id,
            'selector': style.selector,
            'friendly_name': style.friendly_name,
            'styles': style.styles,
            'scope': style.scope
        })
    
    return render(request, 'admin/visual_editor.html', {
        'grouped': grouped,
        'current_scope': scope_filter,
        'title': 'Editor Visual Completo'
    })


@csrf_exempt
@staff_member_required
@require_POST
def save_element_style(request):
    """Guarda o crea estilos de un elemento basado en su selector CSS"""
    try:
        data = json.loads(request.body)
        selector = data.get('selector', '')
        styles = data.get('styles', {})
        friendly_name = data.get('friendly_name', selector)
        scope = data.get('scope', 'global')
        category = data.get('category', 'layout')
        
        if not selector:
            return JsonResponse({'success': False, 'error': 'Selector vacío'}, status=400)
        
        # ✅ Crear o actualizar el estilo basado en el selector
        style_obj, created = CustomStyle.objects.update_or_create(
            selector=selector,
            defaults={
                'friendly_name': friendly_name,
                'styles': styles,
                'scope': scope,
                'category': category
            }
        )
        
        return JsonResponse({
            'success': True,
            'created': created,
            'message': 'Estilo creado' if created else 'Estilo actualizado'
        })
    except Exception as e:
        return JsonResponse({'success': False, 'error': str(e)}, status=500)

@csrf_exempt
@staff_member_required
@require_POST
def save_theme(request):
    """Guarda la configuración del tema"""
    try:
        data = json.loads(request.body)
        mode = data.get('mode', 'light')
        colors = data.get('colors', {})
        
        config = SiteConfig.objects.first()
        if not config:
            config = SiteConfig.objects.create()
        
        # Actualizar colores según el modo
        if mode == 'dark':
            config.dark_primary_color = colors.get('primary', config.dark_primary_color)
            config.dark_secondary_color = colors.get('secondary', config.dark_secondary_color)
            config.dark_background_color = colors.get('background', config.dark_background_color)
            config.dark_text_color = colors.get('text', config.dark_text_color)
        else:
            config.light_primary_color = colors.get('primary', config.light_primary_color)
            config.light_secondary_color = colors.get('secondary', config.light_secondary_color)
            config.light_background_color = colors.get('background', config.light_background_color)
            config.light_text_color = colors.get('text', config.light_text_color)
        
        config.save()
        
        return JsonResponse({'success': True, 'message': 'Tema guardado'})
    except Exception as e:
        return JsonResponse({'success': False, 'error': str(e)}, status=500)

    <script>
        function toggleTheme() {
            const html = document.documentElement;
            const currentTheme = html.getAttribute('data-theme') || 'light';
            const newTheme = currentTheme === 'light' ? 'dark' : 'light';
            
            html.setAttribute('data-theme', newTheme);
            localStorage.setItem('is_dark_mode', newTheme === 'dark');
            document.cookie = `theme_mode=${newTheme};path=/;max-age=31536000`;
        }

        (function initTheme() {
            const savedTheme = localStorage.getItem('is_dark_mode');
            const currentTheme = '{{ current_theme|default:"light" }}';
            let themeToApply = 'light';
            
            if (savedTheme !== null) {
                themeToApply = savedTheme === 'true' ? 'dark' : 'light';
            } else {
                themeToApply = currentTheme;
            }
            document.documentElement.setAttribute('data-theme', themeToApply);
        })();
    </script>

<script>
let currentElement = null;
let liveStyles = {};

function toggleLiveEditor() {
    document.body.classList.toggle('live-editor-mode');
    document.getElementById('live-editor-panel').classList.toggle('open');
    
    if (document.body.classList.contains('live-editor-mode')) {
        document.getElementById('live-editor-toggle').style.display = 'none';
    } else {
        document.getElementById('live-editor-toggle').style.display = 'flex';
        currentElement = null;
    }
}

// Detectar clic en elementos
document.addEventListener('click', function(e) {
    if (!document.body.classList.contains('live-editor-mode')) return;
    if (e.target.id === 'live-editor-toggle' || e.target.closest('#live-editor-panel')) return;
    
    e.preventDefault();
    e.stopPropagation();
    
    currentElement = e.target;
    openEditorForElement(currentElement);
}, true);

function openEditorForElement(el) {
    const computed = window.getComputedStyle(el);
    const selector = generateSelector(el);
    
    // Guardar referencia
    liveStyles = {
        selector: selector,
        background: rgbToHex(computed.backgroundColor) || '#ffffff',
        color: rgbToHex(computed.color) || '#000000',
        padding: computed.padding,
        margin: computed.margin,
        borderRadius: computed.borderRadius,
        fontSize: computed.fontSize,
        fontWeight: computed.fontWeight
    };

    // Renderizar panel
    const panel = document.getElementById('editor-properties');
    panel.innerHTML = `
        <div style="background:#f0f0f0; padding:10px; border-radius:4px; margin-bottom:15px; font-size:12px; word-break:break-all;">
            <strong>Elemento:</strong> ${el.tagName.toLowerCase()}<br>
            <strong>Selector:</strong> <code>${selector}</code>
        </div>

        <div class="editor-input-group">
            <label>Color de Fondo</label>
            <input type="color" value="${liveStyles.background}" oninput="updateStyle('background', this.value)">
        </div>
        <div class="editor-input-group">
            <label>Color de Texto</label>
            <input type="color" value="${liveStyles.color}" oninput="updateStyle('color', this.value)">
        </div>
        <div class="editor-input-group">
            <label>Padding (ej: 10px 20px)</label>
            <input type="text" value="${liveStyles.padding}" oninput="updateStyle('padding', this.value)">
        </div>
        <div class="editor-input-group">
            <label>Margin (ej: 10px 0)</label>
            <input type="text" value="${liveStyles.margin}" oninput="updateStyle('margin', this.value)">
        </div>
        <div class="editor-input-group">
            <label>Border Radius (ej: 8px)</label>
            <input type="text" value="${liveStyles.borderRadius}" oninput="updateStyle('borderRadius', this.value)">
        </div>
        <div class="editor-input-group">
            <label>Tamaño de Fuente (ej: 16px)</label>
            <input type="text" value="${liveStyles.fontSize}" oninput="updateStyle('fontSize', this.value)">
        </div>
    `;
}

function updateStyle(property, value) {
    if (!currentElement) return;
    liveStyles[property] = value;
    
    // Aplicar en tiempo real (camelCase para JS)
    const jsProperty = property.replace(/-([a-z])/g, function (g) { return g[1].toUpperCase(); });
    currentElement.style[jsProperty] = value;
}

function saveLiveStyles() {
    if (!liveStyles.selector) return;

    // Mapear de camelCase a kebab-case para CSS
    const cssStyles = {};
    for (const [key, value] of Object.entries(liveStyles)) {
        if (key !== 'selector' && value) {
            const cssKey = key.replace(/([A-Z])/g, "-$1").toLowerCase();
            cssStyles[cssKey] = value;
        }
    }

    fetch('/api/save-style/', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': getCookie('csrftoken')
        },
        body: JSON.stringify({
            selector: liveStyles.selector,
            friendly_name: liveStyles.selector,
            scope: 'global',
            category: 'layout',
            styles: cssStyles
        })
    })
    .then(res => res.json())
    .then(data => {
        if (data.success) {
            alert('✅ Estilos guardados correctamente');
            toggleLiveEditor();
        } else {
            alert('❌ Error: ' + data.error);
        }
    });
}

// Helpers
function generateSelector(el) {
    if (el.id) return `#${el.id}`;
    if (el.className) {
        const classes = el.className.split(' ').filter(c => c && !c.startsWith('live-editor')).join('.');
        if (classes) return `${el.tagName.toLowerCase()}.${classes}`;
    }
    return el.tagName.toLowerCase();
}

function rgbToHex(rgb) {
    if (!rgb || rgb === 'rgba(0, 0, 0, 0)' || rgb === 'transparent') return null;
    if (rgb.startsWith('#')) return rgb;
    const result = rgb.match(/\d+/g);
    if (!result || result.length < 3) return null;
    return '#' + result.slice(0, 3).map(x => {
        const hex = parseInt(x).toString(16);
        return hex.length === 1 ? '0' + hex : hex;
    }).join('');
}

function getCookie(name) {
    let cookieValue = null;
    if (document.cookie && document.cookie !== '') {
        const cookies = document.cookie.split(';');
        for (let i = 0; i < cookies.length; i++) {
            const cookie = cookies[i].trim();
            if (cookie.substring(0, name.length + 1) === (name + '=')) {
                cookieValue = decodeURIComponent(cookie.substring(name.length + 1));
                break;
            }
        }
    }
    return cookieValue;
}
